-- Convenção adotada: VARIANTE_MINIMAX representa minimax puro, VARIANTE_ALFA_BETA representa poda alfa-beta. Esta convenção será usada dentro da função escolherJogada(estado, variante, randomizador)

-- Na função escolherJogada, o parâmetro randomizador deve receber SORTEIO_DESLIGADO ou SORTEIO_LIGADO, para desativar/ativar o sorteio da lista de listaCandidatos

-- A validade de variante e randomizador, e a condição de raiz não terminal, são verificadas por verificarChamada(estado, variante, randomizador) no início de escolherJogada (ver verificarChamada.txt). A busca só executa se o retorno for RESULTADO_SUCESSO

-- alfa e beta convencionados como -INFINITO e +INFINITO, respectivamente, para ficarem inicialmente fora do intervalo de valores possíveis do tabuleiro

-- JOGADA é a estrutura que contém uma jogada com lance (tipo Move), score (preenchido por converterScore), tipo de score (se é mate, centipeão ou movimento inválido) e resultado da chamada (sucesso, bug ou partida terminada)
-- JOGADA.tipo deve ser TIPO_MATE (representando xeque-mate), TIPO_CENTIPEAO (representando centipeão) ou TIPO_INVALIDO (representando movimento inválido)
-- JOGADA.resultado recebe um dos valores RESULTADO_* definidos em verificarChamada.txt
-- O chamador de escolherJogada deve conferir JOGADA.resultado antes de usar os demais campos. Sempre que JOGADA.resultado != RESULTADO_SUCESSO, os demais campos ficam nos valores padrão (LANCE_NULO, SCORE_INVALIDO, TIPO_INVALIDO), para que toda a estrutura acompanhe o estado da jogada
-- Os valores de JOGADA são válidos após o término de uma chamada de escolherJogada(estado, variante, randomizador)

VARIANTE_MINIMAX = 0
VARIANTE_ALFA_BETA = 1

ALFA = -INFINITO
BETA = INFINITO

LANCE_NULO = 0 -- valor padrão de JOGADA.lance quando não há jogada (representa a1->a1, movimento inválido)
SCORE_INVALIDO = 30000 -- valor padrão de JOGADA.score quando não há jogada. Será int32_t nesta implementação em C

TIPO_INVALIDO = 0 -- valor padrão de JOGADA.tipo quando não há jogada
TIPO_CENTIPEAO = 1
TIPO_MATE = -1

LIMIAR_MATE = MATE - profMax

SORTEIO_LIGADO = 1
SORTEIO_DESLIGADO = 0

função escolherJogada(estado, variante, randomizador):
    listaCandidatos.limpar()
    JOGADA.lance = LANCE_NULO
    JOGADA.score = SCORE_INVALIDO
    JOGADA.tipo = TIPO_INVALIDO
    JOGADA.resultado = verificarChamada(estado, variante, randomizador)

    se JOGADA.resultado != RESULTADO_SUCESSO então
        retorna

    scoreBruto = SCORE_INVALIDO
    se variante == VARIANTE_MINIMAX então
        scoreBruto = minimax(estado, 0)
    senão se variante == VARIANTE_ALFA_BETA então
        scoreBruto = podaAlfaBeta(estado, 0, ALFA, BETA)

    se listaCandidatos != lista vazia então:
        converterScore(scoreBruto, estado.side_to_move, JOGADA) -- ver converterScore.txt

        se randomizador == SORTEIO_LIGADO então
            JOGADA.lance = listaCandidatos[random(0, tamanho(listaCandidatos) - 1)] -- Supõe intervalo fechado nos dois extremos
        senão:
            JOGADA.lance = listaCandidatos[0]

    -- Bloco defensivo: com resultado == RESULTADO_SUCESSO, a raiz tem ao menos um lance e todo score está estritamente entre -INFINITO e +INFINITO, então listaCandidatos nunca fica vazia. Se ficar, é bug interno
    se JOGADA.lance == LANCE_NULO então
        JOGADA.tipo = TIPO_INVALIDO
        JOGADA.score = SCORE_INVALIDO
        JOGADA.resultado = RESULTADO_ERRO_INTERNO
