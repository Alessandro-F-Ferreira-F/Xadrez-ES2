-- testeEmpateRegra(estado) é a função que testa se o estado atual to tabuleiro resulta em um empate
-- gerarLancesLegais(estado) é a função que gera todos os lances válidos, dado um estado
-- emXeque(estado) é a função que testa se o rei da peça da vez está em xeque, dado um estado
-- profMax é a variável global que estipula a profundidade máxima do minimax, dado que a pilha não é infinita e o tempo é limitado

função minimax(estado, profundidade):
    se testeEmpateRegra(estado) então
        retorna empate
    senão se profMax == profundidade então
        retorna avaliar(estado)
    senão 
        listaGerada = gerarLancesLegais(estado)
        se listaGerada == vazio então
            se emXeque(estado) então retorna derrota do jogador da vez
            senão retorna empate
        senão:
            se jogador == X faça
		melhor = -3
		para todo movimento possível faça
			marcar(estado, X, posição)
			atual = minimax(estado, o, profundidade + 1)
			se atual > melhor então 
				melhor = atual
				se profundidade == 0 então
					listaCandidatos.limpar()
					listaCandidatos.adiciona(posição)
			senão se atual == melhor e profundidade == 0 então:
				listaCandidatos.adiciona(posição)
			desmarcar(estado, posição) 
		retornar melhor
	senão:
		melhor = 3
		para todo movimento possível faça
			marcar(estado, O, posição)
			atual = minimax(estado, x, profundidade + 1)
			se atual < melhor então
				melhor = atual
				se profundidade == 0 então
					listaCandidatos.limpar()
					listaCandidatos.adiciona(posição)
			senão se atual == melhor e profundidade == 0 então:
				listaCandidatos.adiciona(posição)
			desmarcar(estado, posição)
		retornar melhor
